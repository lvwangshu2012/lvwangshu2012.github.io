    图论¶

    日期：2026-07-21 参考：李煜东《算法竞赛进阶指南》

    第一章 图的存储¶

    1.1 邻接矩阵¶

    使用一个二维数组 g[i][j] 存储从点 i 到点 j 的边权。

    int g[N][N];
// 加边：g[x][y] = z;
    • 优点：查询两点之间是否有边 O(1)
    • 缺点：空间复杂度 O(n²)，仅适用于 n 较小的稠密图

    1.2 邻接表¶

    每个顶点存储一个链表（或 vector），记录它的所有出边。

    vector 实现：

    vector<pair<int, int>> g[N];
// 加边：g[u].push_back({v, w});

    数组模拟链表实现（进阶指南风格）：

    int ver[M], edge[M], Next[M], head[N], tot;

void add(int x, int y, int z) {
    ver[++tot] = y;   // 边的终点
    edge[tot] = z;    // 边权
    Next[tot] = head[x]; // 下一条边的编号
    head[x] = tot;    // x 的第一条边更新为当前边
}

// 遍历 x 的所有出边
for (int i = head[x]; i; i = Next[i]) {
    int y = ver[i], z = edge[i];
    // ...
}
    • 优点：空间复杂度 O(m)，适用于稀疏图
    • 缺点：查询两点间是否有边 O(deg(u))

    经验：当 m ≈ n² 时为稠密图，用邻接矩阵；否则为稀疏图，用邻接表。

    第二章 最短路¶

    2.1 Dijkstra 算法¶

    基本思想：贪心策略。维护一个"已经确定最短距离的顶点集合"，每次从剩余顶点中选出距离最小的那个，加入集合，并用它松弛所有出边。

    适用条件：所有边的边权非负。

    正确性证明（简述）：由于边权非负，每次选出的距离最小的顶点，其距离不可能再被其他顶点松弛得更小，因此可以永久标记。

    朴素实现（邻接矩阵）¶

    int g[N][N], dist[N];
bool vis[N];

void dijkstra(int s, int n) {
    memset(dist, 0x3f, sizeof dist);
    memset(vis, 0, sizeof vis);
    dist[s] = 0;
    for (int i = 1; i <= n; i++) {
        // 选出距离最小的未标记点
        int t = -1;
        for (int j = 1; j <= n; j++)
            if (!vis[j] && (t == -1 || dist[j] < dist[t]))
                t = j;
        vis[t] = true;
        // 用 t 松弛所有点
        for (int j = 1; j <= n; j++)
            dist[j] = min(dist[j], dist[t] + g[t][j]);
    }
}

    时间复杂度 O(n²)，适用于稠密图。

    堆优化（邻接表）¶

    使用优先队列（小根堆）快速找到距离最小的点。

    using ll = long long;
using pll = pair<ll, ll>;

const int N = 1e6 + 5;

ll n, m;
ll ver[N * 2], nxt[N * 2], head[N * 2], edg[N * 2], tot;

ll d[N];
bool v[N];

void add (ll x, ll y, ll z) {
    ver[++ tot] = y;
    nxt[tot] = head[x];
    edg[tot] = z;
    head[x] = tot;
}

void dijkstra(ll be) {
    memset(d, 0x3f, sizeof(d));
    memset(pre, 0, sizeof(pre));

    priority_queue<pll> pq;
    pq.push({ 0, be });

    d[be] = 0;

    while (!pq.empty()) {
        int x = pq.top().second;

        pq.pop();
        if (v[x]) continue;
        v[x] = true;

        for (int i = head[x]; i; i = nxt[i]) {
            ll y = ver[i];
            ll z = edg[i];

            if (d[y] > d[x] + z) {
                d[y] = d[x] + z;
                pq.push({ -d[y], y });
            }
        }
    }
}

    时间复杂度 O(m log n)，适用于稀疏图。

    注意：稠密图（m 接近 n²）时，堆优化版退化为 O(n² log n)，反而比朴素版 O(n²) 慢。不要只会写堆优化。

    2.2 Bellman-Ford 与 SPFA¶

    Bellman-Ford 算法¶

    基本思想：动态规划。对所有边进行 n-1 轮松弛，第 k 轮后得到经过不超过 k 条边的最短路。

    struct Edge { int x, y, z; } e[M];
int dist[N];

void bellman_ford(int s, int n, int m) {
    memset(dist, 0x3f, sizeof dist);
    dist[s] = 0;
    for (int i = 1; i < n; i++) {      // n-1 轮
        for (int j = 1; j <= m; j++) {  // 每条边
            int x = e[j].x, y = e[j].y, z = e[j].z;
            dist[y] = min(dist[y], dist[x] + z);
        }
    }
}

    时间复杂度 O(nm)。

    判负环：如果第 n 轮仍然能松弛，说明存在负环（最短路最多 n-1 条边）。

    SPFA（队列优化）¶

    基本思想：只有被松弛过的顶点才可能松弛它的邻居，因此用队列维护待扩展的顶点。

    int ver[M], edge[M], Next[M], head[N], tot;
int dist[N];
bool inq[N];  // 是否在队列中
int cnt[N];   // 入队次数，判负环

void add(int x, int y, int z) {
    ver[++tot] = y, edge[tot] = z, Next[tot] = head[x], head[x] = tot;
}

bool spfa(int s, int n) {
    queue<int> q;
    memset(dist, 0x3f, sizeof dist);
    memset(inq, 0, sizeof inq);
    memset(cnt, 0, sizeof cnt);
    dist[s] = 0;
    q.push(s);
    inq[s] = true;
    cnt[s] = 1;
    while (!q.empty()) {
        int x = q.front(); q.pop();
        inq[x] = false;
        for (int i = head[x]; i; i = Next[i]) {
            int y = ver[i], z = edge[i];
            if (dist[y] > dist[x] + z) {
                dist[y] = dist[x] + z;
                if (!inq[y]) {
                    q.push(y);
                    inq[y] = true;
                    if (++cnt[y] >= n) return false; // 存在负环
                }
            }
        }
    }
    return true;
}

    判负环原理：如果一个顶点入队 n 次，说明它被松弛了至少 n 次，而一条最短路最多经过 n-1 条边，因此必然存在负环。

    • 平均时间复杂度 O(m)，最坏 O(nm)（被卡时退化）
    • 有负权边时用 SPFA，无负权边优先 Dijkstra（出题人可能会卡 SPFA）

    2.3 Floyd 算法¶

    基本思想：动态规划。

    设 d[k][i][j] 表示"经过编号不超过 k 的顶点作为中间点"时，从 i 到 j 的最短路径长度。

    阶段划分：k 是阶段，表示允许使用前 k 个顶点作为中转。

    状态转移方程：

    \[d[k][i][j] = \min(d[k-1][i][j],\ d[k-1][i][k] + d[k-1][k][j])\]

    含义： - 不经过第 k 个顶点：d[k-1][i][j] - 经过第 k 个顶点：d[k-1][i][k] + d[k-1][k][j]（i→...→k→...→j）

    两者取最小值。

    不优化版本（三维数组，原始 DP）¶

    int d[N][N][N]; // d[k][i][j] 表示经过前 k 个中间点，i 到 j 的最短路

void floyd(int n) {
    // 初始化：d[0][i][j] = g[i][j]（初始邻接矩阵）
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            d[0][i][j] = g[i][j];

    for (int k = 1; k <= n; k++) {          // 阶段：加入第 k 个中间点
        for (int i = 1; i <= n; i++) {      // 起点
            for (int j = 1; j <= n; j++) {  // 终点
                d[k][i][j] = min(d[k-1][i][j], d[k-1][i][k] + d[k-1][k][j]);
            }
        }
    }
}

    三维版本最直观，严格对应 DP 状态定义，但空间复杂度 O(n³)，n 较大时无法使用。

    空间优化版本（二维数组滚动）¶

    观察状态转移方程：第 k 个阶段只依赖第 k-1 个阶段。

    进一步发现：更新 d[i][j] 时，用到的 d[i][k] 和 d[k][j] 在第 k 轮中不会被修改（因为 d[i][k] = min(d[i][k], d[i][k] + d[k][k])，而 d[k][k] = 0）。因此可以用二维数组原地更新。

    int d[N][N];

void floyd(int n) {
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
}

    空间复杂度从 O(n³) 降到 O(n²)。

    为什么 k 必须在最外层？¶

    k 是阶段，必须从小到大枚举。计算 d[i][j] 时，d[i][k] 和 d[k][j] 必须都是"只经过前 k-1 个中间点"的结果。如果把 k 放在内层，就不能保证这一点。

    类比 01 背包：外层是阶段（物品种类），内层是状态（体积）。Floyd 的 k 就相当于阶段。

    初始化注意¶

    // 初始化：自己到自己距离为 0，其余为 INF
for (int i = 1; i <= n; i++)
    for (int j = 1; j <= n; j++)
        d[i][j] = (i == j) ? 0 : INF;

    扩展应用¶

    1. 传递闭包（连通性判定）

    只关心两点是否连通，不关心距离。用位运算/布尔值替代数值。

    bool d[N][N];

void floyd(int n) {
    for (int k = 1; k <= n; k++)
        for (int i = 1; i <= n; i++)
            for (int j = 1; j <= n; j++)
                d[i][j] |= d[i][k] & d[k][j];
}

    2. 无向图最小环

    在 Floyd 的过程中顺便求经过顶点 k 的最小环。

    int ans = INF;
for (int k = 1; k <= n; k++) {
    // 先算经过 k 的最小环：i → ... → k → j → ... → i
    // 此时 d[i][j] 还是"只经过前 k-1 个中间点"的结果
    for (int i = 1; i < k; i++)
        for (int j = i + 1; j < k; j++)
            ans = min(ans, d[i][j] + g[j][k] + g[k][i]);
    // 再做 Floyd 更新
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= n; j++)
            d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
}

    2.4 分层图最短路¶

    核心思想¶

    当题目中存在某种"可以使用若干次的特殊操作"时（如免费乘车、反向走边、闪现传送等），把状态拆成多层。

    本质是把"使用了几次特殊操作"这个维度加入状态空间，在更大的状态图上跑最短路。

    建模方法¶

    把每个顶点复制 k+1 份，第 t 层表示"已经使用了 t 次特殊操作"。

    • 层内边：普通边，同一层内正常连接，边权不变
    • 层间边：使用一次特殊操作，从第 t 层连向第 t+1 层，边权根据题意变化

    最终答案为 min{ dist[t][终点] }（0 ≤ t ≤ k）。

    例子：免费走 k 条边¶

    // dist[t][u] 表示用了 t 次免费机会，到达 u 的最小花费
// 对每条边 u→v，权值 w：
//   层内：u(t) → v(t)，权 w（不使用免费）
//   层间：u(t) → v(t+1)，权 0（使用一次免费）

    常见应用场景¶

    • 免费走 k 条边 / 半价 k 次
    • 可以反向走 k 次边
    • 传送门 / 闪现 k 次
    • 状态切换（白天/黑夜、装备等级等）

    为什么分层图是对的？ 它把"状态"扩充到了图的维度里。原本只有"在哪个顶点"一个维度，现在加上了"用了几次操作"，形成更大的状态空间。在这个状态空间上跑最短路，就自动枚举了所有使用特殊操作的方式。

    经典例题：P5122 [USACO18DEC] Fine Dining G¶

    题意：N 个牧场，M 条无向小路，所有奶牛都要去牧场 N（牛棚）。有 K 个干草捆，第 i 个在牧场 \(idx_i\)，美味值 \(y_i\)。每头奶牛可以选择在回牛棚的路上停留吃一个干草捆，但要求"经过干草捆后回牛棚的总时间 - 直接回牛棚的最短时间 ≤ 干草捆的美味值"。问每个牧场的奶牛能否做到。

    分层图建模：

    两层图（k=1，因为只能吃一次干草）：

    • 第 0 层：还没吃过干草，编号 1 ~ n
    • 第 1 层：已经吃过干草了，编号 n+1 ~ 2n

    建边方式：

    1. 层内边：每条无向小路在两层各连一遍（层内正常走） - a ↔ b，权 t（第 0 层） - a+n ↔ b+n，权 t（第 1 层）

    2. 层间边：在干草捆处，可以从第 0 层进入第 1 层，表示"在这里吃了干草" - idx → idx+n，权 -y（吃干草获得美味值，相当于减少时间消耗）

    答案判断：

    从牛棚 n 出发跑最短路（因为层间边权为负，用 SPFA）。对于牧场 i 的奶牛： - d[i] = 从 n 到 i 的最短时间（第 0 层，没吃过草）= i 到牛棚的最短时间 - d[i+n] = 从 n 出发，经过"吃一次干草"后，到 i 的最短时间（第 1 层）

    如果 d[i+n] ≤ d[i]，输出 1，否则输出 0。

    为什么从 n 出发？ 无向图中 i→n 等于 n→i。所有奶牛都去 n，从 n 出发跑一遍就能得到所有点到 n 的距离。

    为什么层间边权是 -y？为什么 d[i+n] ≤ d[i] 就是可行？

    题目要求：吃干草的总时间 - 直接回家时间 ≤ 美味值 y

    即：dist(i, idx) + dist(idx, n) - dist[i] ≤ y

    移项得：dist(i, idx) + dist(idx, n) - y ≤ dist[i]

    左边就是"从 i 出发，到 idx 吃草（获得 -y 的'补贴'），再到 n"的时间。

    在分层图中，d[i+n] = min_{所有干草捆 idx} (dist(n, idx) - y + dist(idx, i)) （无向图中 dist(n, idx) = dist(idx, n)，dist(idx, i) = dist(i, idx)）

    所以 d[i+n] ≤ d[i] 就等价于"存在一个干草捆使得增加的时间不超过美味值"。

    AC code：

    #include <bits/stdc++.h>
using namespace std;

using ll = long long;

const int N = 5e4 + 5;
const int M = 1e6 + 5;
const ll INF = 1e18;

int ver[M], nxt[M], head[N * 2];
ll edg[M];
ll d[2 * N];
bool v[2 * N];

int n, m, k, tot;

void add(int x, int y, ll z) {
    ver[++tot] = y;
    edg[tot] = z;
    nxt[tot] = head[x];
    head[x] = tot;
}

void spfa() {
    fill(d, d + 2 * N, INF);
    d[n] = 0;
    v[n] = 1;

    queue<int> q;
    q.push(n);

    while (!q.empty()) {
        int x = q.front();
        q.pop();

        v[x] = 0;
        for (int i = head[x]; i; i = nxt[i]) {
            int y = ver[i];
            ll z = edg[i];
            if (d[y] > d[x] + z) {
                d[y] = d[x] + z;
                if (!v[y]) q.push(y), v[y] = 1;
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> m >> k;
    for (int i = 1; i <= m; i++) {
        int a, b;
        ll t;
        cin >> a >> b >> t;
        add(a, b, t);
        add(b, a, t);
        add(a + n, b + n, t);
        add(b + n, a + n, t);
    }

    for (int i = 1; i <= k; i++) {
        int idx;
        ll y;
        cin >> idx >> y;
        add(idx, idx + n, -y); 
    }

    spfa();

    for (int i = 1; i < n; i++) {
        if (d[i + n] <= d[i])
            cout << "1\n";
        else
            cout << "0\n";
    }

    return 0;
}

    第三章 最小生成树¶

    3.1 Kruskal 算法¶

    基本思想：贪心 + 并查集。将所有边按权值从小到大排序，依次枚举每条边，如果这条边连接的两个顶点不在同一个连通块中，就把它加入最小生成树。

    struct Edge { int x, y, z; } e[M];
int fa[N];

bool cmp(Edge a, Edge b) { return a.z < b.z; }

int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

int kruskal(int n, int m) {
    sort(e + 1, e + m + 1, cmp);
    for (int i = 1; i <= n; i++) fa[i] = i;
    int ans = 0, cnt = 0;
    for (int i = 1; i <= m; i++) {
        int x = find(e[i].x), y = find(e[i].y);
        if (x != y) {
            fa[x] = y;
            ans += e[i].z;
            if (++cnt == n - 1) break;
        }
    }
    return cnt == n - 1 ? ans : -1; // -1 表示图不连通
}

    时间复杂度 O(m log m)，适用于稀疏图。

    稠密图（m ≈ n²）时，排序的 O(n² log n) 不如 Prim 朴素版的 O(n²)。

    3.2 Prim 算法¶

    基本思想：从一个顶点出发，维护"当前生成树到其余各顶点的最短距离"，每次选距离最小的顶点加入生成树，然后更新其余顶点到生成树的距离。

    朴素版（邻接矩阵）¶

    int g[N][N];
int dist[N]; // dist[i] 表示 i 到当前生成树的最小距离
bool vis[N];

int prim(int n) {
    memset(dist, 0x3f, sizeof dist);
    memset(vis, 0, sizeof vis);
    dist[1] = 0;
    int ans = 0;
    for (int i = 1; i <= n; i++) {
        // 找距离生成树最近的未加入顶点
        int t = -1;
        for (int j = 1; j <= n; j++)
            if (!vis[j] && (t == -1 || dist[j] < dist[t]))
                t = j;
        if (dist[t] == 0x3f3f3f3f) return -1; // 不连通
        vis[t] = true;
        ans += dist[t];
        // 更新其它顶点到生成树的距离
        for (int j = 1; j <= n; j++)
            if (!vis[j])
                dist[j] = min(dist[j], g[t][j]);
    }
    return ans;
}

    时间复杂度 O(n²)，适用于稠密图。

    Prim 与 Dijkstra 的对比¶

    算法 dist[i] 的含义 更新方式
    Dijkstra 到起点的距离 dist[t] + g[t][j]
    Prim 到生成树的距离 min(dist[j], g[t][j])

    两者框架非常相似，但核心含义不同。Dijkstra 是累积路径长度，Prim 是取单点到树的最小边。

    3.3 Boruvka 算法（重点）¶

    基本思想：初始时每个顶点是一个连通块。每一轮中，每个连通块选出一条连向外部的最小权边，将两个连通块合并。重复此过程，直到整个图连通。

    特点： - 每一轮连通块的数量至少减半 → 总共最多 log n 轮 - 每一轮遍历所有边 O(m) - 总时间复杂度 O(m log n)

    为什么 Boruvka 很重要？¶

    1. 思想基础："每个连通块选最小出边"是许多高级最小生成树问题的核心思路
    2. 特殊图高效：对于完全图但边权由公式给出的情形（如 w = |a_i - a_j|），Boruvka 配合数据结构可以做到非常优秀的复杂度
    3. 天然并行：各连通块独立选边，天然支持并行计算

    算法流程¶

    初始：每个顶点一个连通块
while 连通块数量 > 1:
    清空每个连通块的最小出边记录
    遍历所有边：
        若两端属于不同连通块：
            用这条边分别更新两个连通块的最小出边
    对每个连通块：
        若它有最小出边且另一端尚未合并：
            合并两个连通块，累加边权

    实现¶

    struct Edge { int x, y, z; } e[M];
int fa[N];
int best[N]; // best[i] = 连通块 i 的最小出边的编号

int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

int boruvka(int n, int m) {
    for (int i = 1; i <= n; i++) fa[i] = i;
    int ans = 0, cnt = n; // cnt = 当前连通块数量

    while (cnt > 1) {
        memset(best, 0, sizeof best); // 0 表示没有出边

        // 遍历所有边，更新各连通块的最小出边
        for (int i = 1; i <= m; i++) {
            int x = find(e[i].x), y = find(e[i].y);
            if (x == y) continue; // 同一连通块，跳过
            if (!best[x] || e[i].z < e[best[x]].z) best[x] = i;
            if (!best[y] || e[i].z < e[best[y]].z) best[y] = i;
        }

        bool merged = false;
        for (int i = 1; i <= n; i++) {
            if (fa[i] != i || !best[i]) continue; // 只处理代表元
            int x = find(e[best[i]].x), y = find(e[best[i]].y);
            if (x != y) {
                fa[x] = y;
                ans += e[best[i]].z;
                cnt--;
                merged = true;
            }
        }
        if (!merged) return -1; // 无法合并，说明不连通
    }
    return ans;
}

    去重技巧：A 的最小出边连向 B，B 的最小出边也连向 A，这条边不能加两次。用并查集 find 之后判断 x != y 再合并，自然避免了重复。

    典型应用¶

    • 完全图的 MST：边权由公式给出（如 \(|a_i - a_j|、a_i XOR a_j\) 等），不需要枚举所有边，每轮对每个点找最近的异块点即可
    • 最小生成树计数、次小生成树等进阶问题

    第四章 拓扑排序¶

    4.1 基本概念¶

    对一个有向无环图（DAG）的顶点进行排序，使得对于每一条有向边 u→v，起点 u 都排在终点 v 的前面。

    基本结论：一个有向图存在拓扑序，当且仅当它是 DAG（没有环）。

    4.2 Kahn 算法（BFS 实现）¶

    基本思想：不断找入度为 0 的顶点，加入拓扑序列，然后删去它的所有出边（即邻居的入度减一）。

    int ver[M], Next[M], head[N], tot;
int ind[N]; // 入度
int ans[N], cnt; // 拓扑序列

void add(int x, int y) {
    ver[++tot] = y, Next[tot] = head[x], head[x] = tot;
    ind[y]++;
}

bool toposort(int n) {
    queue<int> q;
    for (int i = 1; i <= n; i++)
        if (ind[i] == 0) q.push(i);

    while (!q.empty()) {
        int x = q.front(); q.pop();
        ans[++cnt] = x;
        for (int i = head[x]; i; i = Next[i]) {
            int y = ver[i];
            if (--ind[y] == 0) q.push(y);
        }
    }
    return cnt == n; // 全部入队则无环，否则有环
}

    判环：最终拓扑序列长度 == n 则无环，小于 n 则有环（环上的顶点入度永远不会变成 0）。

    4.3 DFS 实现¶

    基本思想：深度优先搜索，回溯时记录顶点，最后逆序就是拓扑序。

    int ver[M], Next[M], head[N], tot;
int vis[N]; // 0=未访问, 1=正在访问, 2=已访问
int ans[N], cnt;

void add(int x, int y) {
    ver[++tot] = y, Next[tot] = head[x], head[x] = tot;
}

bool dfs(int x) {
    vis[x] = 1;
    for (int i = head[x]; i; i = Next[i]) {
        int y = ver[i];
        if (vis[y] == 1) return false; // 发现环
        if (vis[y] == 0 && !dfs(y)) return false;
    }
    vis[x] = 2;
    ans[++cnt] = x; // 回溯时加入
    return true;
}

bool toposort(int n) {
    cnt = 0;
    memset(vis, 0, sizeof vis);
    for (int i = 1; i <= n; i++)
        if (vis[i] == 0 && !dfs(i)) return false;
    reverse(ans + 1, ans + cnt + 1); // 逆序后才是拓扑序
    return true;
}

    4.4 常见应用¶

    1. 判环：判断有向图是否存在环
    2. DAG 上的动态规划：按拓扑序转移，保证无后效性（如最长路、路径方案数等）
    3. 关键路径（工程调度问题）
    4. 依赖关系问题（课程安排、编译顺序等）

    总结对比表¶

    算法 时间复杂度 核心思想 适用场景
    Dijkstra 朴素 O(n²) 贪心选最小 稠密图最短路
    Dijkstra 堆优化 O(m log n) 堆加速选最小 稀疏图最短路
    SPFA O(m) ~ O(nm) 队列优化 Bellman-Ford 负权边、判负环
    Floyd O(n³) DP，k 为中间阶段 全源最短路、n 较小
    Kruskal O(m log m) 排序 + 并查集 稀疏图 MST
    Prim 朴素 O(n²) 贪心加点 稠密图 MST
    Boruvka O(m log n) 每块选最小出边 特殊图 MST、思想重要
    拓扑排序 O(n + m) BFS/DFS 消入度 DAG 判环、DP 序

    核心要点¶

    1. 稠密图用朴素版：Dijkstra 和 Prim 的朴素版在稠密图比堆优化/Kruskal 更快，不要只会写堆优化
    2. Floyd 的 k 是阶段：k 在最外层，类比 01 背包的阶段划分
    3. 分层图：处理"特殊操作"的万能方法，把状态拆成多层
    4. Boruvka：重点掌握思想——每个连通块选最小出边，log n 轮合并
    5. 拓扑排序：DAG 上一切问题的基础，先拓扑序再 DP
